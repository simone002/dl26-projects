\section{Introduzione}

La \textit{segmentazione temporale delle azioni}
(\textit{Temporal Action Segmentation}, TAS)
consiste nel predire, frame per frame, quale azione umana
\`e in corso all'interno di un video lungo e non strutturato.
Formalmente, dato un video $V = \{f_1, \ldots, f_T\}$,
l'obiettivo \`e produrre $\hat{Y} = \{\hat{y}_1, \ldots, \hat{y}_T\}$
dove $\hat{y}_t \in \{0, 1, \ldots, C\}$
e la classe 0 indica assenza di azione (\textit{background}).

A differenza del riconoscimento di azioni su clip brevi,
la TAS richiede predizione densa e localizzazione precisa dei confini.
Le sfide principali sono:
\begin{itemize}
  \item Forte sbilanciamento background/foreground
  \item Variabilit\`a delle durate delle azioni
  \item Confini temporali ambigui
  \item Alta similarit\`a inter-classe (es.\ Open vs Close fridge)
\end{itemize}

\subsection{Obiettivi}
\begin{enumerate}
  \item Implementare e confrontare CNN 1D, LSTM, xLSTM e Mamba.
  \item Addestrare su EGTEA Gaze+ con metriche standard.
  \item Analizzare qualitativamente gli errori sistematici.
\end{enumerate}

\subsection{Feature Pre-estratte con TSN}

Sono state utilizzate feature pre-estratte con TSN
(\textit{Temporal Segment Networks}) \cite{wang2016temporal},
seguendo il protocollo RULSTM \cite{furnari2019would}.
Ogni frame \`e rappresentato da un vettore di 1024 dimensioni (RGB).
Le feature sono memorizzate in LMDB con chiavi
\texttt{VideoSession\_frame\_XXXXXXXXXX.jpg}
e valori di 4096 byte (1024 float32).

\subsection{Costruzione delle Label Dense}

Le annotazioni in millisecondi sono convertite in frame:
\begin{equation}
  f = \left\lfloor \frac{t_{\text{ms}}}{1000} \cdot 24 \right\rfloor
\end{equation}
I frame non annotati ricevono label 0 (background).
Il background \`e incluso perch\'e definisce i confini temporali
tra azioni consecutive.

\begin{algorithm}[H]
\caption{Costruzione label dense per un clip}
\begin{algorithmic}[1]
\State labels $\leftarrow \mathbf{0}^T$
\For{ogni $(a_s, a_e, \text{class})$ in annotazioni[video]}
  \State $o_s \leftarrow \max(a_s, \text{frame\_start})$,
         $o_e \leftarrow \min(a_e, \text{frame\_end})$
  \If{$o_s \leq o_e$}
    \State labels[$o_s{-}\text{start}:o_e{-}\text{start}{+}1$] $\leftarrow$ class
  \EndIf
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{CNN 1D (Baseline)}
Convoluzioni 1D senza memoria temporale.
Ogni layer: Conv1D (kernel 3) + BatchNorm + ReLU + Dropout(0.5).
Receptive field $r=5$ frame. Classificazione con Conv1D $1\times1$.

\subsection{LSTM}
Implementata in modalit\`a bidirezionale \cite{hochreiter1997long}:
$\mathbf{h}_t = [\overrightarrow{\mathbf{h}}_t \,\|\, \overleftarrow{\mathbf{h}}_t]
\in \mathbb{R}^{512}$.
Gate forget sigmoide causa decadimento della memoria per sequenze lunghe.

\subsection{xLSTM}
xLSTM \cite{beck2024xlstm} introduce:
\begin{itemize}
  \item \textbf{sLSTM}: gate esponenziali
        $\tilde{f}_t = e^{\mathbf{w}_f\cdot\mathbf{x}_t}$,
        reset pi\`u netti ai confini delle azioni.
  \item \textbf{mLSTM}: memoria matriciale
        $\mathbf{C}_t = f_t\mathbf{C}_{t-1} + i_t(\mathbf{v}_t\otimes\mathbf{k}_t)$,
        cattura relazioni tra feature nel tempo.
\end{itemize}
Libreria ufficiale \texttt{NX-AI/xlstm}, variante mLSTM a 4 teste.

\subsection{Mamba}
Mamba \cite{gu2023mamba} usa Selective State Space Models:
\begin{equation}
  \mathbf{h}_t = \bar{\mathbf{A}}\mathbf{h}_{t-1} + \bar{\mathbf{B}}x_t,
  \quad y_t = \mathbf{C}\mathbf{h}_t + Dx_t
\end{equation}
con $\Delta_t = \text{softplus}(\mathbf{W}_\Delta x_t)$ dipendente dall'input.
Implementato in PyTorch puro per compatibilit\`a Windows.
Trainato con batch\_size=64 e lr=0.0004 (square root scaling).

\section{Metodologia di Training}
\label{sec:metodologia}

\subsection{Loss Composta}

\begin{equation}
  \mathcal{L} = \mathcal{L}_{\text{CE}} + 0.2\,\mathcal{L}_{\text{Smooth}}
  + 0.3\,\mathcal{L}_{\text{Boundary}}
\end{equation}

\paragraph{CrossEntropy pesata}
$w_0 = 0.05$ (background), $w_c = 1.0$ (foreground), label smoothing $\epsilon=0.05$.

\paragraph{Smooth Loss}
Penalizza transizioni brusche solo dove il GT non cambia etichetta
(ignora i confini reali):
\begin{equation}
  \mathcal{L}_{\text{Smooth}} = \frac{1}{N}
  \sum_{t=1}^{T-1} \mathbf{1}[y_t = y_{t-1}]
  \cdot \|\mathbf{p}_t - \mathbf{p}_{t-1}\|^2
\end{equation}
Riduce la frammentazione delle predizioni senza penalizzare
i confini reali delle azioni. Peso 0.2.

\paragraph{Boundary Loss}
Penalizza gli errori nei frame vicini ai confini con finestra espansa $\pm3$:
\begin{equation}
  \mathcal{L}_{\text{Boundary}} = \text{CE}\bigl(
  \hat{Y}[\mathcal{B}],\; Y[\mathcal{B}]\bigr)
\end{equation}
dove $\mathcal{B} = \{t : |y_t - y_{t-1}| > 0\} \oplus [-3, +3]$
\`e la maschera di confine espansa di $\pm3$ frame.
Fornisce pi\`u esempi di training sui confini, rendendo il modello
pi\`u robusto alla localizzazione temporale. Peso 0.3.

Nota: la boundary loss \`e una CE parziale sui frame di confine
--- tecnicamente un double counting con la CE globale,
ma questa scelta \`e consapevole: si vuole enfatizzare
la precisione temporale sui confini.

\subsection{Ottimizzatore}
AdamW \cite{loshchilov2017decoupled},
CosineAnnealingLR ($T_{\max}=50$, $\eta_{\min}=10^{-6}$),
gradient clipping $\|\nabla\|_2 \leq 1.0$.
LR derivato dalla square root scaling rule \cite{goyal2017accurate}.

\subsection{Metriche}

\paragraph{mIoU} Accumulata epoch-level su 105 classi foreground:
$\text{mIoU} = \frac{1}{105}\sum_{c=1}^{105}
\frac{\text{TP}_c}{\text{TP}_c+\text{FP}_c+\text{FN}_c}$

\paragraph{F1@k} Metrica segment-level: TP se IoU temporale $\geq k$.
Calcolata per $k \in \{10\%, 25\%, 50\%\}$.

\paragraph{Edit Score} Distanza di Levenshtein tra sequenze di segmenti:
$\text{Edit} = 1 - \frac{d_{\text{Lev}}(S_{\text{pred}}, S_{\text{gt}})}
{\max(|S_{\text{pred}}|, |S_{\text{gt}}|)}$

\paragraph{Boundary F1} F1 sui confini temporali con tolleranza 2 frame.

\paragraph{acc\_fg} Accuratezza solo sui frame foreground.

\paragraph{CNN1D e LSTM}
Entrambi ritardano l'inizio: senza contesto globale (CNN1D)
o per accumulo ricorrente (LSTM), il modello necessita di pi\`u frame
per classificare l'azione. La fine viene anticipata per decadimento
della rappresentazione locale/ricorrente.

\paragraph{xLSTM}
Anticipa l'inizio ($-5.9$, exponential gating reattivo)
ma anticipa anche la fine ($-9.3$), meno preciso di Mamba.

\paragraph{Mamba}
Contrariamente all'analisi precedente su 200 clip,
i dati completi (2022 clip) mostrano un ritardo di $+14.1$ frame all'inizio
e un forte anticipo di $-15.5$ frame alla fine.
Il selective scan di Mamba, pur essendo reattivo,
ha bisogno di accumulare evidenza su pi\`u frame
prima di commutare lo stato --- comportamento simile alla LSTM
ma con maggiore varianza per l'assenza di bidirezionalit\`a.

(Figura~\ref{fig:clip_lstm5}):

mentre non rileva l'unica azione reale (\textit{Pour condiment},

\section{Conclusioni}
\label{sec:conclusioni}

\subsection{Contributi}

\begin{enumerate}
  \item Pipeline completa con loss CE+Smooth+Boundary, sliding window,
        metriche mIoU/F1@k/Edit Score/Boundary F1.
  \item Confronto sistematico di 4 architetture sullo stesso setup.
  \item Ablation study: CE+Smooth+Boundary migliora F1@50 del 28\%
        rispetto a CE+Dice.
  \item Analisi qualitativa: pattern distinti per architettura,
        falsi positivi sistematici su background documentati.
\end{enumerate}

\paragraph{Risultato principale}
LSTM \`e il modello migliore (mIoU 0.050, F1@50 0.055, Edit 0.086).
Tutti i modelli tra 3.6--5.0\% mIoU: il bottleneck \`e il setup,
non l'architettura.

\subsection{Limitazioni}
Gap con SOTA (F1@50: 5.5\% vs 48.8\% ASFormer) per:
assenza raffinamento multi-stage, feature TSN vs I3D,
processing per clip invece che per video.

\subsection{Sviluppi Futuri}
\begin{itemize}
  \item Soft-NMS per raffinare i confini predetti.
  \item Processing video-level per contesto globale.
  \item Multi-stage refinement (MS-TCN).
  \item Feature I3D per catturare il movimento.
  \item Cross-validation a 3 split per risultati definitivi.
\end{itemize}

\section{Introduzione}

La \textit{segmentazione temporale delle azioni}
(\textit{Temporal Action Segmentation}, TAS)
consiste nel predire, frame per frame, quale azione umana
\`e in corso all'interno di un video lungo e non strutturato.
Formalmente, dato un video $V = \{f_1, \ldots, f_T\}$,
l'obiettivo \`e produrre $\hat{Y} = \{\hat{y}_1, \ldots, \hat{y}_T\}$
dove $\hat{y}_t \in \{0, 1, \ldots, C\}$
e la classe 0 indica assenza di azione (\textit{background}).

A differenza del riconoscimento di azioni su clip brevi,
la TAS richiede predizione densa e localizzazione precisa dei confini.
Le sfide principali sono:
\begin{itemize}
  \item Forte sbilanciamento background/foreground
  \item Variabilit\`a delle durate delle azioni
  \item Confini temporali ambigui
  \item Alta similarit\`a inter-classe (es.\ Open vs Close fridge)
\end{itemize}

\subsection{Obiettivi}
\begin{enumerate}
  \item Implementare e confrontare CNN 1D, LSTM, xLSTM e Mamba.
  \item Addestrare su EGTEA Gaze+ con metriche standard.
  \item Analizzare qualitativamente gli errori sistematici.
\end{enumerate}

\subsection{Feature Pre-estratte con TSN}

Sono state utilizzate feature pre-estratte con TSN
(\textit{Temporal Segment Networks}) \cite{wang2016temporal},
seguendo il protocollo RULSTM \cite{furnari2019would}.
Ogni frame \`e rappresentato da un vettore di 1024 dimensioni (RGB).
Le feature sono memorizzate in LMDB con chiavi
\texttt{VideoSession\_frame\_XXXXXXXXXX.jpg}
e valori di 4096 byte (1024 float32).

\subsection{Costruzione delle Label Dense}

Le annotazioni in millisecondi sono convertite in frame:
\begin{equation}
  f = \left\lfloor \frac{t_{\text{ms}}}{1000} \cdot 24 \right\rfloor
\end{equation}
I frame non annotati ricevono label 0 (background).
Il background \`e incluso perch\'e definisce i confini temporali
tra azioni consecutive.

\begin{algorithm}[H]
\caption{Costruzione label dense per un clip}
\begin{algorithmic}[1]
\State labels $\leftarrow \mathbf{0}^T$
\For{ogni $(a_s, a_e, \text{class})$ in annotazioni[video]}
  \State $o_s \leftarrow \max(a_s, \text{frame\_start})$,
         $o_e \leftarrow \min(a_e, \text{frame\_end})$
  \If{$o_s \leq o_e$}
    \State labels[$o_s{-}\text{start}:o_e{-}\text{start}{+}1$] $\leftarrow$ class
  \EndIf
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{CNN 1D (Baseline)}
Convoluzioni 1D senza memoria temporale.
Ogni layer: Conv1D (kernel 3) + BatchNorm + ReLU + Dropout(0.5).
Receptive field $r=5$ frame. Classificazione con Conv1D $1\times1$.

\subsection{LSTM}
Implementata in modalit\`a bidirezionale \cite{hochreiter1997long}:
$\mathbf{h}_t = [\overrightarrow{\mathbf{h}}_t \,\|\, \overleftarrow{\mathbf{h}}_t]
\in \mathbb{R}^{512}$.
Gate forget sigmoide causa decadimento della memoria per sequenze lunghe.

\subsection{xLSTM}
xLSTM \cite{beck2024xlstm} introduce:
\begin{itemize}
  \item \textbf{sLSTM}: gate esponenziali
        $\tilde{f}_t = e^{\mathbf{w}_f\cdot\mathbf{x}_t}$,
        reset pi\`u netti ai confini delle azioni.
  \item \textbf{mLSTM}: memoria matriciale
        $\mathbf{C}_t = f_t\mathbf{C}_{t-1} + i_t(\mathbf{v}_t\otimes\mathbf{k}_t)$,
        cattura relazioni tra feature nel tempo.
\end{itemize}
Libreria ufficiale \texttt{NX-AI/xlstm}, variante mLSTM a 4 teste.

\subsection{Mamba}
Mamba \cite{gu2023mamba} usa Selective State Space Models:
\begin{equation}
  \mathbf{h}_t = \bar{\mathbf{A}}\mathbf{h}_{t-1} + \bar{\mathbf{B}}x_t,
  \quad y_t = \mathbf{C}\mathbf{h}_t + Dx_t
\end{equation}
con $\Delta_t = \text{softplus}(\mathbf{W}_\Delta x_t)$ dipendente dall'input.
Implementato in PyTorch puro per compatibilit\`a Windows.
Trainato con batch\_size=64 e lr=0.0004 (square root scaling).

\section{Metodologia di Training}
\label{sec:metodologia}

\subsection{Loss Composta}

\begin{equation}
  \mathcal{L} = \mathcal{L}_{\text{CE}} + 0.2\,\mathcal{L}_{\text{Smooth}}
  + 0.3\,\mathcal{L}_{\text{Boundary}}
\end{equation}

\paragraph{CrossEntropy pesata}
$w_0 = 0.05$ (background), $w_c = 1.0$ (foreground), label smoothing $\epsilon=0.05$.

\paragraph{Smooth Loss}
Penalizza transizioni brusche solo dove il GT non cambia etichetta
(ignora i confini reali):
\begin{equation}
  \mathcal{L}_{\text{Smooth}} = \frac{1}{N}
  \sum_{t=1}^{T-1} \mathbf{1}[y_t = y_{t-1}]
  \cdot \|\mathbf{p}_t - \mathbf{p}_{t-1}\|^2
\end{equation}
Riduce la frammentazione delle predizioni senza penalizzare
i confini reali delle azioni. Peso 0.2.

\paragraph{Boundary Loss}
Penalizza gli errori nei frame vicini ai confini con finestra espansa $\pm3$:
\begin{equation}
  \mathcal{L}_{\text{Boundary}} = \text{CE}\bigl(
  \hat{Y}[\mathcal{B}],\; Y[\mathcal{B}]\bigr)
\end{equation}
dove $\mathcal{B} = \{t : |y_t - y_{t-1}| > 0\} \oplus [-3, +3]$
\`e la maschera di confine espansa di $\pm3$ frame.
Fornisce pi\`u esempi di training sui confini, rendendo il modello
pi\`u robusto alla localizzazione temporale. Peso 0.3.

Nota: la boundary loss \`e una CE parziale sui frame di confine
--- tecnicamente un double counting con la CE globale,
ma questa scelta \`e consapevole: si vuole enfatizzare
la precisione temporale sui confini.

\subsection{Ottimizzatore}
AdamW \cite{loshchilov2017decoupled},
CosineAnnealingLR ($T_{\max}=50$, $\eta_{\min}=10^{-6}$),
gradient clipping $\|\nabla\|_2 \leq 1.0$.
LR derivato dalla square root scaling rule \cite{goyal2017accurate}.

\subsection{Metriche}

\paragraph{mIoU} Accumulata epoch-level su 105 classi foreground:
$\text{mIoU} = \frac{1}{105}\sum_{c=1}^{105}
\frac{\text{TP}_c}{\text{TP}_c+\text{FP}_c+\text{FN}_c}$

\paragraph{F1@k} Metrica segment-level: TP se IoU temporale $\geq k$.
Calcolata per $k \in \{10\%, 25\%, 50\%\}$.

\paragraph{Edit Score} Distanza di Levenshtein tra sequenze di segmenti:
$\text{Edit} = 1 - \frac{d_{\text{Lev}}(S_{\text{pred}}, S_{\text{gt}})}
{\max(|S_{\text{pred}}|, |S_{\text{gt}}|)}$

\paragraph{Boundary F1} F1 sui confini temporali con tolleranza 2 frame.

\paragraph{acc\_fg} Accuratezza solo sui frame foreground.

per l'errore di inizio, confermando il comportamento anticipatorio
dell'exponential gating.

\subsection{Falsi Positivi su Background}

Il 6.4\% dei clip di test (130 clip) \`e di puro background.
Su questi clip tutti i modelli producono \textbf{falsi positivi sistematici}:
predendo azioni inesistenti basandosi sull'apparenza visiva dei frame.

Un esempio emblematico \`e il Clip 5 della LSTM
(Figura~\ref{fig:clip_lstm5}):
il modello predice una sequenza frenetica di azioni diverse
(\textit{Turn on faucet}, \textit{Crack egg}, \textit{Take pan},
\textit{Operate stove}) nei frame 0--62 dove c'\`e solo background,
mentre non rileva l'unica azione reale (\textit{Pour condiment},
frame 62--80).

\begin{figure}[H]
\centering
% \includegraphics[width=\textwidth]{figures/lstm_clip5.png}
\caption{Clip 5 --- LSTM: falsi positivi su background puro
         e mancato riconoscimento di \textit{Pour condiment}.}

Questo comportamento riflette due problemi strutturali:
(1) il bias verso classi frequenti nel training set
(le feature TSN in zone di background assomigliano ad azioni di cucina
perch\'e l'ambiente \`e lo stesso);
(2) l'incapacit\`a di distinguere background ricco di movimento
da azioni reali senza contesto temporale globale.

\section{Conclusioni}
\label{sec:conclusioni}

\subsection{Contributi}

\begin{enumerate}
  \item Pipeline completa con loss CE+Smooth+Boundary, sliding window,
        metriche mIoU/F1@k/Edit Score/Boundary F1.
  \item Confronto sistematico di 4 architetture sullo stesso setup.
  \item Ablation study: CE+Smooth+Boundary migliora F1@50 del 28\%
        rispetto a CE+Dice.
  \item Analisi qualitativa: pattern distinti per architettura,
        falsi positivi sistematici su background documentati.
\end{enumerate}

\paragraph{Risultato principale}
LSTM \`e il modello migliore (mIoU 0.050, F1@50 0.055, Edit 0.086).
Tutti i modelli tra 3.6--5.0\% mIoU: il bottleneck \`e il setup,
non l'architettura.

\subsection{Limitazioni}
Gap con SOTA (F1@50: 5.5\% vs 48.8\% ASFormer) per:
assenza raffinamento multi-stage, feature TSN vs I3D,
processing per clip invece che per video.

\subsection{Sviluppi Futuri}
\begin{itemize}
  \item Soft-NMS per raffinare i confini predetti.
  \item Processing video-level per contesto globale.
  \item Multi-stage refinement (MS-TCN).
  \item Feature I3D per catturare il movimento.
  \item Cross-validation a 3 split per risultati definitivi.
\end{itemize}
